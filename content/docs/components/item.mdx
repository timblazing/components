## Installation

```npm
npx shadcn@latest add item
```

## Usage

```tsx
import {
  Item,
  ItemActions,
  ItemContent,
  ItemDescription,
  ItemGroup,
  ItemMedia,
  ItemSeparator,
  ItemTitle,
} from "@/components/ui/item"
```

```tsx
<Item variant="outline">
  <ItemContent>
    <ItemTitle>Title</ItemTitle>
    <ItemDescription>Description</ItemDescription>
  </ItemContent>
  <ItemActions>
    <Button size="sm">Action</Button>
  </ItemActions>
</Item>
```

## Guidelines

- Use `Item` for list rows and settings; use a [Table](/components/table) for tabular data.
- Keep descriptions to one or two lines; they are clamped.
- Put at most one primary action in `ItemActions`.
