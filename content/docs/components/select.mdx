## Installation

```npm
npx shadcn@latest add select
```

## Usage

```tsx
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from "@/components/ui/select"
```

```tsx
<Select items={fruits}>
  <SelectTrigger className="w-48">
    <SelectValue placeholder="Select a fruit" />
  </SelectTrigger>
  <SelectContent>
    {fruits.map((f) => (
      <SelectItem key={f.value} value={f.value}>
        {f.label}
      </SelectItem>
    ))}
  </SelectContent>
</Select>
```

## Guidelines

- Always pass `items` so the trigger shows readable labels.
- `onValueChange` receives `value | null`. Handle the `null` case.
- Use a [Combobox](/components/combobox) when the list is long enough to need search.
