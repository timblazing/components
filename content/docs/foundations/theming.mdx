<Steps>
<Step>

### Create a shadcn/ui project with Base UI

```npm
npx shadcn@latest init -b base -p nova
```

</Step>
<Step>

### Replace the theme variables

Copy the `:root` and `.dark` blocks from this site's [`global.css`](https://github.com/timblazing/components/blob/main/src/app/global.css) into your stylesheet, along with the extra `@theme inline` mappings:

```css
@theme inline {
  --font-sans: var(--font-geist-sans), system-ui, sans-serif;
  --font-mono: var(--font-geist-mono), ui-monospace, monospace;
  --font-serif: var(--font-geist-serif-regular), ui-serif, serif;
  --font-display: var(--font-serif);
  --font-heading: var(--font-sans);
  --color-success: var(--success);
  --color-warning: var(--warning);
  --color-info: var(--info);
  --color-border-strong: var(--border-strong);
}
```

</Step>
<Step>

### Load the font families

Install the bundled fonts so builds do not depend on a font CDN:

```npm
npm install geist
```

```tsx title="app/layout.tsx"
import { GeistSans } from "geist/font/sans"
import { GeistMono } from "geist/font/mono"
import { GeistSerifRegular } from "geist/font/serif/regular"

export default function RootLayout({ children }: { children: React.ReactNode }) {
  return (
    <html lang="en" className={`${GeistSans.variable} ${GeistMono.variable} ${GeistSerifRegular.variable}`} suppressHydrationWarning>
      <body>{children}</body>
    </html>
  )
}
```

Apply the font roles in your stylesheet:

```css
@layer base {
  html { @apply font-sans; }
  code, kbd, pre { @apply font-mono; }
}

.font-display {
  font-family: var(--font-display);
  font-weight: 400;
  letter-spacing: -0.04em;
}
```

</Step>
<Step>

### Add components

Anything from [Components](/components), [Blocks](/blocks), or [Charts](/charts) now picks up the theme.

```npm
npx shadcn@latest add button card chart
```

</Step>
</Steps>
